# Each pipeline is a directory: the logic in a library so the tests can drive it, and a thin
# main beside it.
function(ssk_pipeline_warnings)
  foreach(target ${ARGV})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
    endif()
    if(SSK_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE $<IF:$<BOOL:${MSVC}>,/WX,-Werror>)
    endif()
  endforeach()
endfunction()

add_library(ssk_corpus corpus.cpp)
target_link_libraries(ssk_corpus PUBLIC ssk)
target_include_directories(ssk_corpus PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
ssk_pipeline_warnings(ssk_corpus)
